AULA 3 • MIGRATION

Schema::create('candidatos', function (Blueprint $table) {
    $table->id();
    $table->string('nome');
    $table->string('numero', 2)->unique();
    $table->string('partido', 80);
    $table->string('foto')->nullable();
    $table->boolean('ativo')->default(true);

AULA 4 • CONTROLLER

Salvando o candidato

public function store(Request $request): RedirectResponse
{
    $dados = $this->validar($request);
    $dados['ativo'] = $request->boolean('ativo');

    if ($request->hasFile('foto')) {
        $dados['foto'] = $request->file('foto')
            ->store('candidatos', 'public');
    }

    Candidato::create($dados);

    return redirect()->route('administracao.candidatos.index')
        ->with('sucesso', 'Candidato cadastrado.');
}

DB::transaction(function () use ($dados) {

    if (! $eleitor || $eleitor->jaVotou()) {
        throw ValidationException::withMessages([...]);
    }

    Voto::create([...]);
    $eleitor->update(['votou_em' => now()]);
});
